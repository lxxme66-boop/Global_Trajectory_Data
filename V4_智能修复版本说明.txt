💎 核心代码
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

if actual_count != expected_count:
    # 情况1：多了 → 取前 N 个
    if actual_count > expected_count:
        scores = scores[:expected_count]  # 简单粗暴，非常有效
        
    # 情况2：少了 → 补充低分
    elif actual_count < expected_count:
        missing = expected_count - actual_count
        scores.extend([-10.0] * missing)  # 低分排最后，不影响结果
